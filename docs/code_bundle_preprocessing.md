```python
"""Safe CSV loading with schema validation. Datasets are read only."""

import pandas as pd

from feedforward.data.schemas import TABLES


class SchemaError(ValueError):
    """Raised when a dataset file does not have the expected columns."""


def load_table(name: str) -> pd.DataFrame:
    """Load one dataset by name, check its columns and parse its date columns."""
    if name not in TABLES:
        raise KeyError(f"Unknown table '{name}'. Known tables: {sorted(TABLES)}")
    spec = TABLES[name]
    if not spec.path.exists():
        raise FileNotFoundError(f"Dataset file not found: {spec.path}")
    df = pd.read_csv(spec.path, encoding="utf-8")
    missing = [c for c in spec.columns if c not in df.columns]
    extra = [c for c in df.columns if c not in spec.columns]
    if missing or extra:
        raise SchemaError(f"{spec.path.name}: missing columns {missing}, unexpected {extra}")
    for col in spec.date_cols:
        df[col] = pd.to_datetime(df[col], errors="raise")
    return df


def load_all() -> dict[str, pd.DataFrame]:
    """Load and validate every dataset table."""
    return {name: load_table(name) for name in TABLES}
```

### `src/feedforward/data/nlp_text.py`

```python
"""Text pipeline for dish names, weather text, event names and typed input.

NLP phases covered here: preprocessing (clean, lowercase, tokenise, normalise) and
fuzzy matching. Every NLTK resource is optional: if it cannot be loaded (offline), a simple
regex based fallback is used so the project still runs.
"""

import re
import unicodedata
from dataclasses import dataclass

from rapidfuzz import fuzz, process

from feedforward import config as C

# ---------------------------------------------------------------- NLTK with offline fallback
_RESOURCES = {
    "punkt_tab": "tokenizers/punkt_tab",
    "wordnet": "corpora/wordnet",
    "averaged_perceptron_tagger_eng": "taggers/averaged_perceptron_tagger_eng",
    "stopwords": "corpora/stopwords",
}
_available: dict[str, bool] = {}


def setup_nltk(download: bool = True) -> dict[str, bool]:
    """Make NLTK resources available from artifacts/nltk_data. Never raises when offline."""
    if _available and not download:
        return _available
    try:
        import nltk
    except ImportError:
        return {name: False for name in _RESOURCES}
    C.NLTK_DATA_DIR.mkdir(parents=True, exist_ok=True)
    if str(C.NLTK_DATA_DIR) not in nltk.data.path:
        nltk.data.path.insert(0, str(C.NLTK_DATA_DIR))
    for name, path in _RESOURCES.items():
        try:
            nltk.data.find(path)
            _available[name] = True
        except LookupError:
            ok = False
            if download:
                try:
                    ok = bool(nltk.download(name, download_dir=str(C.NLTK_DATA_DIR), quiet=True))
                except Exception:  # offline or blocked network
                    ok = False
            _available[name] = ok
    return _available


def has(resource: str) -> bool:
    """True when an NLTK resource can be used."""
    if not _available:
        setup_nltk()
    return _available.get(resource, False)


# ---------------------------------------------------------------- cleaning and normalising
# Spelling variants that neither stemmers nor lemmatisers can fix. Used for MATCHING only.
SPELLING_MAP = {
    "briyani": "biryani", "briyaani": "biryani", "pulav": "pulao", "pulaav": "pulao",
    "chappathi": "chapati", "chapathi": "chapati", "chappati": "chapati",
    "kuruma": "korma", "kurma": "korma", "raitha": "raita", "sabji": "sabzi",
    "idli": "idly", "omelette": "omelet", "vadai": "vada", "mattar": "matar", "daal": "dal",
    "kichadi": "khichdi", "kichdi": "khichdi", "poori": "puri", "tiffen": "tiffin",
    "chilly": "chilli", "veggie": "veg", "vegetable": "veg",
}
ABBREVIATIONS = {"spl": "special", "veg": "veg", "pnr": "paneer"}

_QUOTES = str.maketrans({"‘": "'", "’": "'", "“": '"', "”": '"',
                         "–": "-", "—": "-", " ": " "})
_STRAY = re.compile(r"[^\w\s'%/&.\-+]")  # keep letters, digits, and a few useful marks


def clean_text(text: object) -> str:
    """Remove stray symbols, standardise punctuation and collapse spaces. Keeps case."""
    if text is None or (isinstance(text, float) and text != text):
        return ""
    s = unicodedata.normalize("NFKC", str(text)).translate(_QUOTES)
    s = _STRAY.sub(" ", s)
    s = re.sub(r"\s*([.,;:!?])\s*(?=\s|$)", r"\1", s)  # no space before end punctuation
    return re.sub(r"\s+", " ", s).strip()


def display_name(text: object) -> str:
    """Name for the screen: cleaned, original case kept; all lower case input is title cased."""
    s = clean_text(text)
    return s.title() if s and s == s.lower() else s


def match_key(text: object) -> str:
    """Lower case key for matching: cleaned, abbreviations expanded, spelling variants unified."""
    words = re.findall(r"[a-z0-9]+(?:'[a-z]+)?", clean_text(text).lower())
    words = [ABBREVIATIONS.get(w, w) for w in words]
    return " ".join(SPELLING_MAP.get(w, w) for w in words)


# ---------------------------------------------------------------- tokenisation
def tokenize_whitespace(text: str) -> list[str]:
    """Split on spaces only. Keeps punctuation glued to words."""
    return text.split()


def tokenize_regex(text: str) -> list[str]:
    """Word characters only. Drops punctuation but splits 'Gobi 65' and 'Chef's' badly."""
    return re.findall(r"\w+", text)


def tokenize_word(text: str) -> list[str]:
    """NLTK word tokeniser (falls back to a regex that keeps punctuation as tokens)."""
    if has("punkt_tab"):
        from nltk.tokenize import word_tokenize

        return word_tokenize(text)
    return re.findall(r"\w+(?:'\w+)?|[^\w\s]", text)


def merge_expressions(tokens: list[str], expressions: set[tuple[str, ...]]) -> list[str]:
    """Multi word expression handling: join known phrases into one token with underscores.

    Expressions are given as match keys, so 'Veg Briyani' matches ('veg', 'biryani').
    Longest phrase wins, so a dish stays one entity and not two words.
    """
    longest = max((len(e) for e in expressions), default=1)
    lowered = [SPELLING_MAP.get(t.lower(), t.lower()) for t in tokens]
    out, i = [], 0
    while i < len(tokens):
        for n in range(min(longest, len(tokens) - i), 1, -1):
            if tuple(lowered[i:i + n]) in expressions:
                out.append("_".join(tokens[i:i + n]))
                i += n
                break
        else:
            out.append(tokens[i])
            i += 1
    return out


def ngrams(tokens: list[str], n: int) -> list[tuple[str, ...]]:
    """Word n-grams: n=1 unigram, 2 bigram, 3 trigram."""
    return [tuple(tokens[i:i + n]) for i in range(len(tokens) - n + 1)]


def char_ngrams(text: str, n: int = 3) -> list[str]:
    """Character n-grams (with edge padding), robust to small spelling differences."""
    padded = f" {text} "
    return [padded[i:i + n] for i in range(len(padded) - n + 1)]


# ---------------------------------------------------------------- stop words
_FALLBACK_STOP = {"a", "an", "the", "is", "are", "was", "of", "to", "in", "on", "at", "for", "and",
                  "or", "why", "what", "how", "do", "does", "it", "this", "that", "be", "if",
                  "no", "not", "nor", "than", "too", "very"}
NEGATIONS = {"no", "not", "nor", "never", "don't", "doesn't", "isn't", "won't", "without"}


def stop_words() -> set[str]:
    """English stop words (NLTK list, or a small fallback list)."""
    if has("stopwords"):
        from nltk.corpus import stopwords

        return set(stopwords.words("english"))
    return set(_FALLBACK_STOP)


def remove_stopwords(tokens: list[str], keep_negations: bool = True) -> list[str]:
    """Drop stop words. Use ONLY for intent detection; keep them for entity extraction.

    Negations are kept by default because "no rain" must not become "rain".
    """
    stops = stop_words() - (NEGATIONS if keep_negations else set())
    return [t for t in tokens if t.lower() not in stops]


# ---------------------------------------------------------------- stemming and lemmatising
def stem_porter(word: str) -> str:
    """Porter stemmer: chops suffixes by rules, can produce non words."""
    from nltk.stem import PorterStemmer

    return PorterStemmer().stem(word)


def stem_regexp(word: str) -> str:
    """Regexp stemmer: strips a few simple suffixes."""
    from nltk.stem import RegexpStemmer

    return RegexpStemmer("ing$|s$|es$|ed$", min=4).stem(word)


def lemmatize(word: str) -> str:
    """WordNet lemmatiser (dictionary based). Falls back to stripping a plural s."""
    if has("wordnet"):
        from nltk.stem import WordNetLemmatizer

        try:
            return WordNetLemmatizer().lemmatize(word.lower())
        except LookupError:
            pass
    w = word.lower()
    return w[:-1] if len(w) > 3 and w.endswith("s") and not w.endswith("ss") else w


# ---------------------------------------------------------------- fuzzy matching
@dataclass(frozen=True)
class FuzzyResult:
    """Best catalogue match for a messy name."""

    query: str
    match: str | None
    score: float
    needs_review: bool


def fuzzy_match(query: str, catalogue: dict[str, str], threshold: float = C.FUZZY_ACCEPT,
                review_floor: float = C.FUZZY_REVIEW, scorer=fuzz.WRatio) -> FuzzyResult:
    """Match a messy name to the catalogue (match_key -> id) using rapidfuzz.

    score >= threshold: accepted. review_floor <= score < threshold: accepted but listed for
    human review. Below review_floor: no match.
    """
    key = match_key(query)
    if key in catalogue:
        return FuzzyResult(query, catalogue[key], 100.0, False)
    best = process.extractOne(key, list(catalogue), scorer=scorer)
    if best is None or best[1] < review_floor:
        return FuzzyResult(query, None, best[1] if best else 0.0, True)
    return FuzzyResult(query, catalogue[best[0]], float(best[1]), best[1] < threshold)
```

### `src/feedforward/data/nlp_entities.py`

```python
"""Sentence splitting, POS tagging, rule based NER and word sense rules for typed questions.

All rule and gazetteer based: no model is trained and no LLM is used here. This is the NLU
front end of the copilot (intent plus entities). Stop words are NOT removed before NER.
"""

import re
from dataclasses import dataclass

from rapidfuzz import fuzz

from feedforward.data.nlp_text import (
    clean_text,
    fuzzy_match,
    has,
    match_key,
    stop_words,
    tokenize_word,
)

MEAL_SLOTS = {"breakfast": "breakfast", "lunch": "lunch", "snacks": "snacks", "snack": "snacks",
              "dinner": "dinner", "supper": "dinner", "tiffin": "breakfast"}
WEEKDAYS = ["monday", "tuesday", "wednesday", "thursday", "friday", "saturday", "sunday"]
RELATIVE_DAYS = {"today": 0, "tomorrow": 1}
WEATHER_WORDS = {"rain": "Rain", "rainy": "Rain", "raining": "Rain", "drizzle": "Drizzle",
                 "thunder": "Thunder", "storm": "Thunder", "hot": "Hot", "heat": "Hot",
                 "fog": "Fog", "mist": "Mist", "haze": "Haze", "cloudy": "Partly Cloudy",
                 "dry": "Dry", "sunny": "Fair", "fair": "Fair"}
EVENT_PATTERNS = {
    "exam_week": r"\b(exam|exams|examination|formative test|theory exam|test week)\b",
    "festival": r"\b(festival|deepavali|diwali|pongal|ayudha puja|vijayadashami|christmas)\b",
    "holiday": r"\b(holiday|holidays|vacation|semester break)\b",
}
ABBREV = ["spl", "mr", "mrs", "dr", "vs", "no", "approx", "st"]


@dataclass(frozen=True)
class Entity:
    """One detected entity: type, original text span, normalised value."""

    type: str
    text: str
    value: str | float | None = None


def split_sentences(text: str) -> list[str]:
    """Sentence boundary detection that does not break at abbreviations like 'Spl.'."""
    protected = text
    for ab in ABBREV:
        protected = re.sub(rf"\b({ab})\.", r"\1<DOT>", protected, flags=re.I)
    parts = re.split(r"(?<=[.!?])\s+", protected.strip())
    return [p.replace("<DOT>", ".") for p in parts if p]


def pos_tag(text: str) -> list[tuple[str, str]]:
    """Part of speech tags (NLTK tagger, or a simple number / noun fallback)."""
    tokens = tokenize_word(clean_text(text))
    if has("averaged_perceptron_tagger_eng"):
        import nltk

        return nltk.pos_tag(tokens)
    return [(t, "CD" if re.fullmatch(r"\d+(\.\d+)?%?", t) else "NN") for t in tokens]


def candidate_entities(text: str) -> dict[str, list[str]]:
    """Use POS tags to list candidate dish words (nouns) and numbers (cardinals)."""
    tagged = pos_tag(text)
    return {"nouns": [w for w, t in tagged if t.startswith("NN")],
            "numbers": [w for w, t in tagged if t == "CD"]}


# ---------------------------------------------------------------- word sense rules
_INGREDIENT_CONTEXT = re.compile(r"\b(milk)\b\s+(powder|stock|litres?|liters?|kg|used|usage|"
                                 r"ingredient)|\b(ingredient|stock of)\s+milk")
_ADJ_SWEET = re.compile(r"\b(too|very|less|more|bit|so|extra)\s+sweet\b|\bsweet\s+(tooth|taste|"
                        r"potato|corn|lime)\b")


def sense_of(word: str, text: str) -> str:
    """Small word sense disambiguation: 'milk' and 'sweet' as a dish or something else."""
    low = text.lower()
    if word == "milk":
        return "ingredient" if _INGREDIENT_CONTEXT.search(low) else "dish"
    if word == "sweet":
        return "adjective" if _ADJ_SWEET.search(low) else "dish"
    return "dish"


# ---------------------------------------------------------------- NER
def _dish_entities(text: str, dish_keys: dict[str, str], max_len: int = 5) -> list[Entity]:
    """Gazetteer lookup of dish names over word windows, longest first, spelling tolerant."""
    words = match_key(text).split()
    stops = stop_words()  # only used to stop fuzzy windows starting or ending on a filler word
    found, i = [], 0
    while i < len(words):
        hit = None
        for n in range(min(max_len, len(words) - i), 0, -1):
            phrase = " ".join(words[i:i + n])
            if phrase in dish_keys:
                hit = (n, dish_keys[phrase], phrase)
            elif len(phrase) >= 5 and words[i] not in stops and words[i + n - 1] not in stops:
                res = fuzzy_match(phrase, dish_keys, threshold=88, review_floor=88,
                                  scorer=fuzz.ratio)
                if res.match and not res.needs_review:
                    hit = (n, res.match, phrase)
            if hit:
                break
        near = " ".join(words[max(0, i - 2):i + hit[0] + 2]) if hit else ""
        if hit and sense_of(hit[2], near) == "dish":
            found.append(Entity("DISH", hit[2], hit[1]))
            i += hit[0]
        else:
            i += hit[0] if hit else 1
    return found


def extract_entities(text: str, dish_keys: dict[str, str]) -> list[Entity]:
    """Rule and gazetteer NER: DISH, MEAL_SLOT, WEEKDAY, DATE, WEATHER, EVENT, PERCENT, NUMBER."""
    clean = clean_text(text)
    low = clean.lower()
    ents = _dish_entities(clean, dish_keys)
    words = re.findall(r"[a-z]+", low)
    ents += [Entity("MEAL_SLOT", w, MEAL_SLOTS[w]) for w in words if w in MEAL_SLOTS]
    ents += [Entity("WEEKDAY", w, w) for w in words if w in WEEKDAYS]
    ents += [Entity("DATE", w, RELATIVE_DAYS[w]) for w in words if w in RELATIVE_DAYS]
    ents += [Entity("DATE", m, m) for m in re.findall(r"\b\d{4}-\d{2}-\d{2}\b", low)]
    ents += [Entity("WEATHER", w, WEATHER_WORDS[w]) for w in words if w in WEATHER_WORDS]
    for name, pattern in EVENT_PATTERNS.items():
        m = re.search(pattern, low)
        if m:
            ents.append(Entity("EVENT", m.group(0), name))
    pct = re.findall(r"(-?\d+(?:\.\d+)?)\s*(?:%|percent|per cent)", low)
    ents += [Entity("PERCENT", p, float(p)) for p in pct]
    pct_set = set(pct)
    nums = re.findall(r"(?<![\d-])(-?\d+(?:\.\d+)?)(?!\d|-\d|\s*(?:%|percent|per cent))", low)
    ents += [Entity("NUMBER", n, float(n)) for n in nums if n not in pct_set]
    return ents
```

```python
"""M1 preprocessing: validate, unify types and ids, normalise text, write artifacts/processed/."""

import re

import pandas as pd

from feedforward import config as C
from feedforward.data.loaders import load_all
from feedforward.data.nlp_text import (
    clean_text,
    display_name,
    fuzzy_match,
    lemmatize,
    match_key,
    setup_nltk,
)
from feedforward.data.text_features import DishTextModel

# Weather words -> flags. Condition text is semi structured: one short phrase per day.
WEATHER_FLAGS = {"is_thunder": r"thunder", "is_drizzle": r"drizzle|shower", "is_fog": r"fog|mist|haze|smoke",
                 "is_rain_text": r"rain|drizzle|shower|thunder"}
# Event kinds found in the free text calendar notes.
EVENT_KINDS = {"rain_holiday": r"rain holiday", "compensatory": r"compensatory", "cyclone": r"cyclone",
               "monsoon": r"monsoon", "bank_only": r"bank"}


def normalise_ids(df: pd.DataFrame) -> pd.DataFrame:
    """Unify ids: dish_id upper case and trimmed, ingredient lower case and trimmed."""
    df = df.copy()
    if "dish_id" in df:
        df["dish_id"] = df["dish_id"].astype(str).str.strip().str.upper()
    if "ingredient" in df:
        df["ingredient"] = df["ingredient"].astype(str).map(clean_text).str.lower()
    return df


def dish_catalogue(dishes: pd.DataFrame) -> dict[str, str]:
    """match_key -> dish_id; fails loudly if two dishes share one key after normalising."""
    keys = dishes["dish_name"].map(match_key)
    clash = dishes.loc[keys.duplicated(keep=False), "dish_name"].tolist()
    if clash:
        raise ValueError(f"Dish names collide after normalising: {clash}")
    return dict(zip(keys, dishes["dish_id"], strict=True))


def process_dishes(dishes: pd.DataFrame, model: DishTextModel) -> pd.DataFrame:
    """Add display name, match key, lemma tokens and the text embedding to the catalogue."""
    out = dishes.copy()
    out["dish_name"] = out["dish_name"].map(display_name)
    out["match_key"] = out["dish_name"].map(match_key)
    out["lemmas"] = out["match_key"].map(lambda s: " ".join(lemmatize(w) for w in s.split()))
    return out.merge(model.embedding_table(), on="dish_id", how="left")


def process_weather(weather: pd.DataFrame) -> pd.DataFrame:
    """Normalise condition text, add text flags and reliability flags."""
    w = weather.copy()
    w["condition"] = w["condition"].map(display_name)
    key = w["condition"].map(match_key)
    for col, pattern in WEATHER_FLAGS.items():
        w[col] = key.str.contains(pattern).astype(int)
    w["weather_reliable"] = (~w["data_quality"].str.upper().str.startswith("UNVERIFIED")).astype(int)
    w["temp_outlier"] = w["data_quality"].str.lower().str.contains("outlier").astype(int)
    return w.drop(columns=["rainfall_mm_reported"])  # always 0, not usable


def process_calendar(cal: pd.DataFrame) -> pd.DataFrame:
    """Normalise holiday and event names, add event kind flags from the free text note."""
    c = cal.copy()
    c["holiday_name"] = c["holiday_name"].fillna("").map(display_name)
    note = c["event_note"].fillna("").map(clean_text)
    c["event_note"] = note
    low = note.str.lower()
    for col, pattern in EVENT_KINDS.items():
        c[f"event_{col}"] = low.str.contains(pattern, flags=re.I).astype(int)
    c["phase_estimated"] = (c["phase_confidence"] == "estimated").astype(int)
    return c


def map_kitchen_names(raw: pd.DataFrame, catalogue: dict[str, str]) -> pd.DataFrame:
    """Map every distinct raw dish name to a dish_id. Returns name, id, score, review flag."""
    rows = []
    for name in raw["dish_name"].dropna().unique():
        res = fuzzy_match(name, catalogue)
        rows.append({"raw_name": name, "dish_id": res.match, "score": round(res.score, 1),
                     "needs_review": res.needs_review})
    return pd.DataFrame(rows)


def main() -> None:
    """Run the whole M1 pipeline and write processed tables."""
    setup_nltk()
    C.PROCESSED_DIR.mkdir(parents=True, exist_ok=True)
    t = load_all()
    dishes = normalise_ids(t["dishes"])
    catalogue = dish_catalogue(dishes)
    model = DishTextModel().fit(dishes)
    outputs = {
        "dishes": process_dishes(dishes, model),
        "weather": process_weather(t["weather"]),
        "calendar": process_calendar(t["calendar"]),
        "weekly_menu": normalise_ids(t["weekly_menu"]),
        "recipes": normalise_ids(t["recipes"]),
        "inventory_ingredients": normalise_ids(t["inventory_ingredients"]),
        "inventory_batches": normalise_ids(t["inventory_batches"]),
        "inventory_daily": normalise_ids(t["inventory_daily"]),
        "dish_name_map": map_kitchen_names(t["kitchen_history_raw"], catalogue),
    }
    for name, df in outputs.items():
        df.to_parquet(C.PROCESSED_DIR / f"{name}.parquet", index=False)
        print(f"wrote {name}: {len(df)} rows")
    review = outputs["dish_name_map"].query("needs_review")
    print(f"dish names needing review: {len(review)}")


if __name__ == "__main__":
    main()
```

### `src/feedforward/data/cleaning.py`

```python
"""Clean kitchen_history_raw.csv. The clean file is used only as ground truth to test this.

Steps (each one is counted in the cleaning log):
1. drop exact duplicate rows
2. normalise dish names with the text pipeline and map them to the catalogue
3. repair missing attendance and students_present from other rows of the same date and slot
4. recompute missing or negative leftover_qty as prepared + topup - consumed
5. recompute leftover_kg from portion size when leftover_qty was repaired
6. check consistency and flag outliers (never silently removed)
"""

import json

import pandas as pd

from feedforward import config as C
from feedforward.data.loaders import load_table
from feedforward.data.nlp_text import display_name, fuzzy_match
from feedforward.data.preprocess import dish_catalogue, normalise_ids

OUTLIER_Z = 4.0  # robust z score (median and MAD) above which a quantity is flagged


def _fill_group_value(df: pd.DataFrame, col: str, keys: list[str]) -> int:
    """Fill missing values of col with the value other rows of the same group share."""
    before = int(df[col].isna().sum())
    df[col] = df[col].fillna(df.groupby(keys)[col].transform("max"))
    return before - int(df[col].isna().sum())


def flag_outliers(df: pd.DataFrame) -> pd.Series:
    """Flag unusual demand per attendee for a dish and slot (robust z score; flag only, never drop)."""
    demand = (df["consumed_qty"] + df["unmet_qty"]) / df["attendance"]  # per attendee, so crowds do not count
    grp = demand.groupby([df["meal_slot"], df["dish_id"]])
    med = grp.transform("median")
    mad = (demand - med).abs().groupby([df["meal_slot"], df["dish_id"]]).transform("median")
    z = 0.6745 * (demand - med) / mad.where(mad > 0)
    return (z.abs() > OUTLIER_Z).fillna(False)


def clean_kitchen(raw: pd.DataFrame, dishes: pd.DataFrame) -> tuple[pd.DataFrame, dict]:
    """Return the cleaned table and a log of how many rows each step touched."""
    log: dict[str, int] = {"rows_in": len(raw)}
    df = raw.drop_duplicates().copy()
    log["duplicates_dropped"] = len(raw) - len(df)

    catalogue = dish_catalogue(dishes)
    names = df["dish_name"].map(lambda n: fuzzy_match(n, catalogue))
    log["dish_names_changed"] = int((df["dish_name"] != df["dish_name"].map(display_name)).sum())
    log["dish_names_needing_review"] = int(sum(n.needs_review for n in names))
    df["dish_id"] = [n.match if n.match else d for n, d in zip(names, df["dish_id"].str.upper(), strict=True)]
    names_by_id = dishes.set_index("dish_id")["dish_name"]
    df["dish_name"] = df["dish_id"].map(names_by_id)

    log["attendance_repaired"] = _fill_group_value(df, "attendance", ["date", "meal_slot"])
    log["students_present_repaired"] = _fill_group_value(df, "students_present", ["date"])

    left = df["prepared_qty"] + df["topup_qty"] - df["consumed_qty"]
    bad = df["leftover_qty"].isna() | (df["leftover_qty"] < 0)
    log["leftover_missing"] = int(df["leftover_qty"].isna().sum())
    log["leftover_negative"] = int((df["leftover_qty"] < 0).sum())
    df.loc[bad, "leftover_qty"] = left[bad]
    kg_per_portion = df["dish_id"].map(dishes.set_index("dish_id")["portion_g"]) / 1000
    df.loc[bad, "leftover_kg"] = (df.loc[bad, "leftover_qty"] * kg_per_portion[bad]).round(2)

    for col in ("students_present", "attendance", "leftover_qty"):
        df[col] = df[col].astype(int)
    log["consistency_violations"] = int((left != df["leftover_qty"]).sum())
    df["demand_outlier"] = flag_outliers(df)
    log["demand_outliers_flagged"] = int(df["demand_outlier"].sum())
    log["rows_out"] = len(df)
    return df.sort_values(["date", "meal_slot", "dish_id"]).reset_index(drop=True), log


def compare_with_truth(clean: pd.DataFrame, truth: pd.DataFrame) -> dict:
    """Compare the cleaned table with kitchen_history.csv column by column."""
    keys = ["date", "meal_slot", "dish_id"]
    a = clean.drop(columns="demand_outlier").sort_values(keys).reset_index(drop=True)
    b = truth.sort_values(keys).reset_index(drop=True)
    result = {"same_row_count": len(a) == len(b)}
    if not result["same_row_count"]:
        return result
    diffs = {c: int((a[c] != b[c]).sum()) for c in b.columns if (a[c] != b[c]).any()}
    result["columns_with_differences"] = diffs
    return result


def main() -> None:
    """Clean the raw file, save it with the log and print the ground truth comparison."""
    dishes = normalise_ids(load_table("dishes"))
    clean, log = clean_kitchen(load_table("kitchen_history_raw"), dishes)
    log["ground_truth_comparison"] = compare_with_truth(clean, load_table("kitchen_history"))
    C.PROCESSED_DIR.mkdir(parents=True, exist_ok=True)
    clean.to_parquet(C.PROCESSED_DIR / "kitchen_clean.parquet", index=False)
    C.REPORTS_DIR.mkdir(exist_ok=True)
    (C.REPORTS_DIR / "cleaning_log.json").write_text(json.dumps(log, indent=2, default=str), encoding="utf-8")
    print(json.dumps(log, indent=2, default=str))


if __name__ == "__main__":
    main()
```

```python
"""Feature extraction from text: bag of words, TF-IDF, cosine similarity, SVD embedding."""

import numpy as np
import pandas as pd
from scipy.sparse import hstack
from sklearn.decomposition import TruncatedSVD
from sklearn.feature_extraction.text import CountVectorizer, TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

from feedforward import config as C
from feedforward.data.nlp_text import match_key


class DishTextModel:
    """TF-IDF of dish names (word 1 to 2 grams plus character 2 to 4 grams) and a small embedding.

    The embedding (TF-IDF then SVD) is the hand engineered text feature. The VAE also learns its
    own dish embedding from data; comparing both is part of the report.
    """

    def __init__(self, dim: int = C.DISH_TEXT_DIM):
        self.dim = dim
        self.word = TfidfVectorizer(ngram_range=(1, 2))
        self.char = TfidfVectorizer(analyzer="char_wb", ngram_range=(2, 4))
        self.svd = TruncatedSVD(n_components=dim, random_state=C.SEED)
        self.ids: list[str] = []

    def fit(self, dishes: pd.DataFrame) -> "DishTextModel":
        """Fit on the dish catalogue (columns dish_id, dish_name)."""
        keys = [match_key(n) for n in dishes["dish_name"]]
        self.ids = dishes["dish_id"].tolist()
        self.matrix = hstack([self.word.fit_transform(keys), self.char.fit_transform(keys)]).tocsr()
        self.embedding = self.svd.fit_transform(self.matrix)
        return self

    def _vector(self, text: str):
        key = [match_key(text)]
        return hstack([self.word.transform(key), self.char.transform(key)]).tocsr()

    def most_similar(self, text: str, top: int = 3) -> list[tuple[str, float]]:
        """Cosine similarity of a typed name to every dish; returns (dish_id, score)."""
        sims = cosine_similarity(self._vector(text), self.matrix)[0]
        order = np.argsort(-sims)[:top]
        return [(self.ids[i], float(sims[i])) for i in order]

    def embedding_table(self) -> pd.DataFrame:
        """One row per dish: text_emb_0 .. text_emb_{dim-1}."""
        cols = [f"text_emb_{i}" for i in range(self.dim)]
        return pd.DataFrame(self.embedding, columns=cols).assign(dish_id=self.ids)


def bag_of_words(names: list[str]) -> pd.DataFrame:
    """Plain bag of words counts (shown in the report next to TF-IDF)."""
    cv = CountVectorizer()
    mat = cv.fit_transform([match_key(n) for n in names])
    return pd.DataFrame(mat.toarray(), columns=cv.get_feature_names_out())
```

```python
    conf = pd.concat([senior["conf"], first["conf"], cal["hconf"].fillna("official")], axis=1)
    rank = conf.apply(lambda col: col.map(CONFIDENCE_RANK)).max(axis=1)
    cal["phase_confidence"] = rank.map({v: k for k, v in CONFIDENCE_RANK.items()}).to_numpy()
    both = pd.concat([cal["senior_phase"], cal["first_year_phase"]], axis=1)
    cal["university_exam_any"] = both.isin(EXAM_PHASES).any(axis=1).astype(int)
    cal["formative_test_any"] = both.apply(lambda c: c.str.startswith("formative")).any(axis=1).astype(int)
    cal["students_mostly_away"] = cal["senior_phase"].isin(AWAY_PHASES).astype(int)
    cal["event_note"] = "Future date, seeded from SRM schedules; confidence in phase_confidence"
    cal["source_codes"] = "S2"
    return cal.drop(columns="hconf")


def weather_climatology(weather: pd.DataFrame) -> pd.DataFrame:
    """Monthly average of last year's reliable weather, used when no forecast is available."""
    ok = weather[weather["weather_reliable"] == 1]
    return ok.groupby(ok["date"].dt.month)[["temp_max_c", "temp_min_c", "rain_level"]].mean()


def assumed_weather(dates: pd.DatetimeIndex, weather: pd.DataFrame) -> pd.DataFrame:
    """Weather rows for dates without data: monthly climatology, flagged as assumed."""
    clim = weather_climatology(weather)
    months = pd.Series(dates.month, index=range(len(dates)))
    out = pd.DataFrame({"date": dates})
    out["temp_max_c"] = months.map(clim["temp_max_c"]).to_numpy()
    out["temp_min_c"] = months.map(clim["temp_min_c"]).to_numpy()
    out["temp_mean_c"] = (out["temp_max_c"] + out["temp_min_c"]) / 2
    out["rain_level"] = months.map(clim["rain_level"]).round().to_numpy()
    out["weather_reliable"] = 1
    out["weather_assumed"] = 1
    return out


def main() -> None:
    """Write data_store/calendar_future.csv."""
    cal = build_future_calendar()
    C.DATA_STORE_DIR.mkdir(exist_ok=True)
    cal.to_csv(C.CALENDAR_FUTURE_CSV, index=False, encoding="utf-8")
    print(f"wrote {C.CALENDAR_FUTURE_CSV.name}: {len(cal)} rows, {cal.date.min().date()} to {cal.date.max().date()}")


if __name__ == "__main__":
    main()
```

### `src/feedforward/data/features.py`

```python
"""Feature engineering. All functions work for any date, so forecasting uses the same code.

Hand engineered here: calendar, weather, menu and lag features. The VAE later learns its own
dish embedding, which is the learned counterpart of the text embedding built here.
"""

import numpy as np
import pandas as pd

from feedforward import config as C
from feedforward.data.calendar_future import assumed_weather
from feedforward.data.feature_registry import LAGS, MEAL_SLOTS, PHASE_GROUPS
from feedforward.data.feature_registry import TEXT_DIM as _TEXT_DIM

CAP_DAYS = 60
HOT_C = 36.0
WEATHER_NOISE_C = 1.5       # forecast error added to temperatures during training
RAIN_FLIP_PROB = 0.10       # chance that a forecast rain level is wrong by one step
PHASE_MAP = {
    "classes": "classes", "formative_test_I": "formative", "formative_test_II": "formative",
    "formative_test_III": "formative", "formative_test_IV": "formative",
    "practical_exam": "practical", "arrear_practical": "practical",
    "theory_exam_period": "theory", "study_gap": "study_gap", "semester_break": "away",
    "summer_semester": "away", "enrolment": "enrolment", "orientation": "enrolment",
}
CONFIDENCE_RANK = {"official": 0, "probable": 1, "estimated": 2}
CALENDAR_COLS = ["date", "tn_public_holiday", "major_festival", "special_declared_holiday",
                 "compensatory_working_day", "rain_holiday_colleges", "senior_phase", "first_year_phase",
                 "phase_confidence", "university_exam_any", "formative_test_any", "students_mostly_away"]


def full_calendar() -> pd.DataFrame:
    """Real calendar followed by the seeded future calendar."""
    real = pd.read_parquet(C.PROCESSED_DIR / "calendar.parquet")[CALENDAR_COLS]
    future = pd.read_csv(C.CALENDAR_FUTURE_CSV, parse_dates=["date"], encoding="utf-8")[CALENDAR_COLS]
    return pd.concat([real, future], ignore_index=True).sort_values("date").reset_index(drop=True)


def full_weather(dates: pd.DatetimeIndex) -> pd.DataFrame:
    """Weather for every requested date: real where known, last year's monthly average otherwise."""
    real = pd.read_parquet(C.PROCESSED_DIR / "weather.parquet")
    real = real.assign(weather_assumed=0)
    known = real[real["date"].isin(dates)]
    missing = dates.difference(pd.DatetimeIndex(known["date"]))
    parts = [known]
    if len(missing):
        parts.append(assumed_weather(pd.DatetimeIndex(missing), real))
    cols = ["date", "temp_max_c", "temp_min_c", "temp_mean_c", "rain_level", "weather_reliable", "weather_assumed"]
    return pd.concat(parts)[cols].sort_values("date").reset_index(drop=True)


def _days_to_and_since(is_event: pd.Series, dates: pd.Series) -> tuple[np.ndarray, np.ndarray]:
    """Days until the next event day and since the last one (capped)."""
    event_days = dates[is_event.astype(bool)].to_numpy()
    nxt, since = np.full(len(dates), CAP_DAYS), np.full(len(dates), CAP_DAYS)
    for i, d in enumerate(dates.to_numpy()):
        after = event_days[event_days >= d]
        before = event_days[event_days <= d]
        if len(after):
            nxt[i] = min(CAP_DAYS, (after[0] - d) / np.timedelta64(1, "D"))
        if len(before):
            since[i] = min(CAP_DAYS, (d - before[-1]) / np.timedelta64(1, "D"))
    return nxt, since


def date_features(calendar: pd.DataFrame, weather: pd.DataFrame) -> pd.DataFrame:
    """One row per date with calendar and weather features (the same for any slot or dish)."""
    cal = calendar.copy()
    out = pd.DataFrame({"date": cal["date"]})
    dow, month = cal["date"].dt.dayofweek, cal["date"].dt.month
    out["dow_sin"], out["dow_cos"] = np.sin(2 * np.pi * dow / 7), np.cos(2 * np.pi * dow / 7)
    out["month_sin"], out["month_cos"] = np.sin(2 * np.pi * month / 12), np.cos(2 * np.pi * month / 12)
    out["is_weekend"] = (dow >= 5).astype(int)
    for col in ("tn_public_holiday", "major_festival", "special_declared_holiday",
                "compensatory_working_day", "rain_holiday_colleges", "university_exam_any",
                "formative_test_any", "students_mostly_away"):
        out[col] = cal[col].astype(int)
    holiday = (cal["tn_public_holiday"] | cal["special_declared_holiday"]).astype(int)
    out["days_to_next_holiday"], out["days_since_last_holiday"] = _days_to_and_since(holiday, cal["date"])
    for cohort, col in (("senior", "senior_phase"), ("first", "first_year_phase")):
        group = cal[col].map(PHASE_MAP)
        for g in PHASE_GROUPS:
            out[f"{cohort}_{g}"] = (group == g).astype(int)
    out["phase_confidence_rank"] = cal["phase_confidence"].map(CONFIDENCE_RANK).astype(int)
    out = out.merge(weather, on="date", how="left")
    out["hot_day"] = (out["temp_max_c"] >= HOT_C).astype(int)
    return out


def add_weather_noise(df: pd.DataFrame, rng: np.random.Generator) -> pd.DataFrame:
    """Training only: perturb weather like a real forecast, so the model does not trust it fully."""
    df = df.copy()
    shift = rng.normal(0, WEATHER_NOISE_C, len(df))
    for col in ("temp_max_c", "temp_min_c", "temp_mean_c"):
        df[col] = df[col] + shift
    df["hot_day"] = (df["temp_max_c"] >= HOT_C).astype(int)
    flip = rng.random(len(df)) < RAIN_FLIP_PROB
    step = rng.choice([-1, 1], len(df))
    df["rain_level"] = np.where(flip, np.clip(df["rain_level"] + step, 0, 3), df["rain_level"])
    return df


def menu_rows_for(dates: pd.DatetimeIndex, weekly_menu: pd.DataFrame) -> pd.DataFrame:
    """Dishes served on each date, from the fixed weekly menu (briyani only on 2nd and 4th Wednesday)."""
    frames = []
    for d in dates:
        nth = (d.day - 1) // 7 + 1
        special_ok = d.dayofweek == 2 and nth in (2, 4)
        rows = weekly_menu[(weekly_menu["day_of_week"] == d.day_name())
                           & ((weekly_menu["condition"] == "always") | special_ok)]
        frames.append(rows.assign(date=d))
    out = pd.concat(frames, ignore_index=True)
    out["is_briyani_special"] = (out["condition"] != "always").astype(int)
    return out[["date", "meal_slot", "dish_id", "is_choice", "n_options", "is_briyani_special"]]


def dish_features(dishes: pd.DataFrame, history: pd.DataFrame) -> pd.DataFrame:
    """One row per dish: catalogue values, codes and the text embedding."""
    out = dishes[["dish_id", "popularity_score", "portion_g"]].copy()
    out["category_code"] = dishes["category"].astype("category").cat.codes
    out["diet_code"] = dishes["diet"].astype("category").cat.codes
    out["is_core"] = (dishes["forecast_priority"] == "core").astype(int)
    limited = history.groupby("dish_id")["special_limited"].max()
    out["special_limited"] = out["dish_id"].map(limited).fillna(0).astype(int)
    emb = [f"text_emb_{i}" for i in range(_TEXT_DIM)]
    out[emb] = dishes[emb].to_numpy()
    out["dish_idx"] = np.arange(len(dishes))
    return out


def _past_table(history: pd.DataFrame) -> pd.DataFrame:
    """Per row quantities that are only used with a lag of at least 7 days."""
    h = history.copy()
    h["demand"] = h["consumed_qty"] + h["unmet_qty"]
    h["rate"] = h["demand"] / h["attendance"]
    h["leftover_rate"] = h["leftover_qty"] / (h["prepared_qty"] + h["topup_qty"]).clip(lower=1)
    h["unmet_rate"] = h["unmet_qty"] / h["demand"].clip(lower=1)
    return h[["date", "meal_slot", "dish_id", "demand", "rate", "leftover_rate", "unmet_rate"]]


def lag_features(history: pd.DataFrame, targets: pd.DataFrame) -> pd.DataFrame:
    """History features for target rows (date, meal_slot, dish_id). Only lags of 7+ days are used."""
    past = _past_table(history)
    slot_att = history.groupby(["date", "meal_slot"], as_index=False)["attendance"].first()
    out = targets[["date", "meal_slot", "dish_id"]].copy()
    for lag in LAGS:
        shifted = past.assign(date=past["date"] + pd.Timedelta(days=lag))
        shifted.columns = ["date", "meal_slot", "dish_id"] + [f"{c}_lag_{lag}" for c in shifted.columns[3:]]
        out = out.merge(shifted, on=["date", "meal_slot", "dish_id"], how="left")
        att = slot_att.assign(date=slot_att["date"] + pd.Timedelta(days=lag)).rename(
            columns={"attendance": f"slot_att_lag_{lag}"})
        out = out.merge(att, on=["date", "meal_slot"], how="left")
        out[f"has_lag_{lag}"] = out[f"demand_lag_{lag}"].notna().astype(int)
    out["n_lags"] = out[[f"has_lag_{n}" for n in LAGS]].sum(axis=1)

    def mean_of(prefix: str) -> pd.Series:
        return out[[f"{prefix}_lag_{n}" for n in LAGS]].mean(axis=1)

    out["demand_mean3"], out["rate_mean3"] = mean_of("demand"), mean_of("rate")
    out["leftover_rate_mean3"], out["unmet_rate_mean3"] = mean_of("leftover_rate"), mean_of("unmet_rate")
    out["slot_att_mean3"] = mean_of("slot_att")
    out["demand_std3"] = out[[f"demand_lag_{n}" for n in LAGS]].std(axis=1)
    out["demand_trend"] = out["demand_lag_7"] - out["demand_lag_14"]
    for lag in LAGS:  # a missing lag is filled with the mean of the lags that exist (else 0)
        for prefix in ("demand", "rate", "slot_att"):
            col = f"{prefix}_lag_{lag}"
            out[col] = out[col].fillna(out[f"{prefix}_mean3"])
    keep = ["demand_std3", "demand_trend", "demand_mean3", "rate_mean3", "leftover_rate_mean3",
            "unmet_rate_mean3", "slot_att_mean3"]
    out[keep] = out[keep].fillna(0)
    lag_cols = [f"{p}_lag_{n}" for p in ("demand", "rate", "slot_att") for n in LAGS]
    out[lag_cols] = out[lag_cols].fillna(0)
    return out.drop(columns=[f"{p}_lag_{n}" for p in ("leftover_rate", "unmet_rate") for n in LAGS])


def build_features(targets: pd.DataFrame, history: pd.DataFrame, dishes: pd.DataFrame) -> pd.DataFrame:
    """All model inputs for target rows (date, meal_slot, dish_id, is_choice, n_options, is_briyani_special)."""
    cal = full_calendar()  # whole calendar, so "days to next holiday" can look past the target dates
    dfeat = date_features(cal, full_weather(pd.DatetimeIndex(cal["date"])))
    out = targets.merge(dfeat, on="date", how="left")
    out = out.merge(dish_features(dishes, history), on="dish_id", how="left")
    for s in MEAL_SLOTS:
        out[f"slot_{s}"] = (out["meal_slot"] == s).astype(int)
    out = out.merge(lag_features(history, targets), on=["date", "meal_slot", "dish_id"], how="left")
    return out
```

```python
"""Chronological splits, rolling origin folds and regimes. Rows are never shuffled."""

import pandas as pd

from feedforward import config as C

REGIMES = ("normal", "exam", "holiday_festival", "vacation", "rainy")


def assign_split(dates: pd.Series) -> pd.Series:
    """Label each date train, val or test by the fixed cut dates in config."""
    out = pd.Series("train", index=dates.index)
    out[(dates >= C.VAL_START) & (dates <= C.VAL_END)] = "val"
    out[(dates >= C.TEST_START) & (dates <= C.TEST_END)] = "test"
    return out


def rolling_origin_folds(n_folds: int = C.N_FOLDS, days: int = C.FOLD_DAYS) -> list[dict]:
    """Expanding window folds: train on everything before each block of `days` days."""
    end = pd.Timestamp(C.HISTORY_END)
    first = end - pd.Timedelta(days=n_folds * days - 1)
    folds = []
    for k in range(n_folds):
        start = first + pd.Timedelta(days=k * days)
        stop = start + pd.Timedelta(days=days - 1)
        folds.append({"fold": k + 1, "train_end": start - pd.Timedelta(days=1), "test_start": start,
                      "test_end": stop})
    return folds


def regime(df: pd.DataFrame) -> pd.Series:
    """Label each row by the situation it is in. Later rules override earlier ones (vacation beats everything)."""
    out = pd.Series("normal", index=df.index)
    out[df["rain_level"] >= 2] = "rainy"
    out[df["university_exam_any"] == 1] = "exam"  # formative test weeks are normal class days
    out[(df["tn_public_holiday"] == 1) | (df["major_festival"] == 1)
        | (df["special_declared_holiday"] == 1)] = "holiday_festival"
    out[df["students_mostly_away"] == 1] = "vacation"
    return out
```

### `src/feedforward/data/timeseries.py`

```python
"""Build the model datasets in artifacts/timeseries/ and the feature registry in reports/.

(a) panel: one row per date, slot and dish with features and the demand target
(b) sequence windows: 21 days of history per dish and slot, for RNN, LSTM or GRU models
(c) slot attendance series
(d) day level totals
"""

import numpy as np
import pandas as pd

from feedforward import config as C
from feedforward.data.cleaning import clean_kitchen
from feedforward.data.feature_registry import BANNED_INPUTS, BANNED_REGISTRY, FEATURE_COLUMNS, LAGS, REGISTRY
from feedforward.data.features import build_features, menu_rows_for
from feedforward.data.loaders import load_table
from feedforward.data.preprocess import normalise_ids
from feedforward.data.splits import assign_split, regime

SEQ_CHANNELS = ["log_demand", "rate", "leftover_rate", "unmet_rate", "served"]
AUDIT_COLS = ["prepared_qty", "topup_qty", "consumed_qty", "unmet_qty", "leftover_qty", "attendance"]


def load_clean_history() -> pd.DataFrame:
    """Cleaned kitchen history (from the raw file) with the demand target added."""
    path = C.PROCESSED_DIR / "kitchen_clean.parquet"
    if path.exists():
        hist = pd.read_parquet(path)
    else:
        hist, _ = clean_kitchen(load_table("kitchen_history_raw"), normalise_ids(load_table("dishes")))
    hist["demand_qty"] = hist["consumed_qty"] + hist["unmet_qty"]
    return hist


def build_panel(history: pd.DataFrame, dishes: pd.DataFrame, menu: pd.DataFrame) -> pd.DataFrame:
    """Panel table: features, target (demand_qty), audit columns, split and regime."""
    dates = pd.DatetimeIndex(sorted(history["date"].unique()))
    targets = menu_rows_for(dates, menu).merge(history[["date", "meal_slot", "dish_id"]],
                                               on=["date", "meal_slot", "dish_id"], how="inner")
    panel = build_features(targets, history, dishes)
    panel = panel.merge(history[["date", "meal_slot", "dish_id", "demand_qty", *AUDIT_COLS]],
                        on=["date", "meal_slot", "dish_id"], how="left")
    panel["split"] = assign_split(panel["date"])
    panel["regime"] = regime(panel)
    return panel.sort_values(["date", "meal_slot", "dish_id"]).reset_index(drop=True)


def build_windows(panel: pd.DataFrame, history: pd.DataFrame) -> tuple[dict, pd.DataFrame]:
    """Sequence windows: 21 days of past channels ending 7 days before the target day."""
    h = history.copy()
    h["log_demand"] = np.log1p(h["demand_qty"])
    h["rate"] = h["demand_qty"] / h["attendance"]
    h["leftover_rate"] = h["leftover_qty"] / (h["prepared_qty"] + h["topup_qty"]).clip(lower=1)
    h["unmet_rate"] = h["unmet_qty"] / h["demand_qty"].clip(lower=1)
    h["served"] = 1.0
    series_id = {k: i for i, k in enumerate(sorted(set(zip(h["meal_slot"], h["dish_id"], strict=True))))}
    start = h["date"].min()
    pad, n_days = C.SEQ_LOOKBACK + min(LAGS), (h["date"].max() - start).days + 1
    grid = np.zeros((len(series_id), n_days + pad, len(SEQ_CHANNELS)), dtype=np.float32)
    sid = [series_id[k] for k in zip(h["meal_slot"], h["dish_id"], strict=True)]
    day = (h["date"] - start).dt.days.to_numpy() + pad
    grid[sid, day] = h[SEQ_CHANNELS].to_numpy(dtype=np.float32)

    p_sid = np.array([series_id[k] for k in zip(panel["meal_slot"], panel["dish_id"], strict=True)])
    p_day = (panel["date"] - start).dt.days.to_numpy() + pad
    first = p_day - min(LAGS) - C.SEQ_LOOKBACK + 1   # window covers t-27 .. t-7
    offsets = np.arange(C.SEQ_LOOKBACK)
    x_seq = grid[p_sid[:, None], first[:, None] + offsets[None, :]]
    arrays = {"x_seq": x_seq, "x_cov": panel[FEATURE_COLUMNS].to_numpy(dtype=np.float32),
              "y": panel["demand_qty"].to_numpy(dtype=np.float32),
              "feature_names": np.array(FEATURE_COLUMNS), "channels": np.array(SEQ_CHANNELS)}
    index = panel[["date", "meal_slot", "dish_id", "dish_idx", "split", "regime"]].copy()
    return arrays, index


def attendance_series(history: pd.DataFrame) -> pd.DataFrame:
    """One row per date and slot: people who ate, plus those present on campus."""
    return history.groupby(["date", "meal_slot"], as_index=False).agg(
        attendance=("attendance", "first"), students_present=("students_present", "first"),
        enrolled=("enrolled", "first"))


def day_totals(history: pd.DataFrame) -> pd.DataFrame:
    """Total demand per date and slot over all dishes (a choice group counts each alternative)."""
    return history.pivot_table(index="date", columns="meal_slot", values="demand_qty", aggfunc="sum").reset_index()


def registry_markdown() -> str:
    """Feature registry as a markdown table (name, group, definition, leakage safe)."""
    rows = ["| Feature | Group | Definition | Leakage safe |", "|---|---|---|---|"]
    rows += [f"| {f.name} | {f.group} | {f.definition} | {'yes' if f.leakage_safe else 'no'} |"
             for f in REGISTRY + BANNED_REGISTRY]
    return "# Feature registry\n\nGenerated by `python -m feedforward.data.timeseries`.\n\n" + "\n".join(rows) + "\n"


def main() -> None:
    """Build and save every dataset."""
    history = load_clean_history()
    dishes = pd.read_parquet(C.PROCESSED_DIR / "dishes.parquet")
    menu = pd.read_parquet(C.PROCESSED_DIR / "weekly_menu.parquet")
    panel = build_panel(history, dishes, menu)
    assert not (set(FEATURE_COLUMNS) & BANNED_INPUTS), "a banned column is in the feature list"
    assert not panel[FEATURE_COLUMNS].isna().any().any(), "NaN in model inputs"
    arrays, index = build_windows(panel, history)
    C.TIMESERIES_DIR.mkdir(parents=True, exist_ok=True)
    panel.to_parquet(C.TIMESERIES_DIR / "panel.parquet", index=False)
    np.savez_compressed(C.TIMESERIES_DIR / "windows.npz", **arrays)
    index.to_parquet(C.TIMESERIES_DIR / "windows_index.parquet", index=False)
    attendance_series(history).to_parquet(C.TIMESERIES_DIR / "attendance_series.parquet", index=False)
    day_totals(history).to_parquet(C.TIMESERIES_DIR / "day_totals.parquet", index=False)
    (C.REPORTS_DIR / "feature_registry.md").write_text(registry_markdown(), encoding="utf-8")
    print(f"panel {panel.shape}, windows {arrays['x_seq'].shape}, features {len(FEATURE_COLUMNS)}")
    print(panel["split"].value_counts().to_dict(), panel["regime"].value_counts().to_dict())


if __name__ == "__main__":
    main()
```
